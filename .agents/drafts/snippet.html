export async function handlePayment(req: Request) { const idempotencyKey = req.headers.get("Idempotency-Key"); if (!idempotencyKey) { return new Response("Missing Idempotency-Key header", { status: 400 }); } // 1. Check Redis / DB if key already exists const existingRecord = await redis.get(`idempotency:${idempotencyKey}`); if (existingRecord) { // Return cached response instantly without re-charging return Response.json(JSON.parse(existingRecord)); } // 2. Lock key with a short TTL (60s) to block concurrent duplicate taps await redis.set(`idempotency:${idempotencyKey}`, JSON.stringify({ status: "processing" }), "EX", 60); // 3. Execute actual business logic safely const result = await processStripeCharge(req); // 4. Cache final response payload for 24 hours await redis.set(`idempotency:${idempotencyKey}`, JSON.stringify(result), "EX", 86400); return Response.json(result); }
